    5.1. مقدمة

    نستعرض هنا التطبيق IMPOTS الذي تم استخدامه مرارًا وتكرارًا في الكتيب الدراسي الخاص بلغة Java لنفس المؤلف. ولنتذكر المشكلة المطروحة. يتعلق الأمر بكتابة تطبيق يسمح بحساب الضريبة المستحقة على أحد دافعي الضرائب. سنأخذ الحالة المبسطة لدافع ضرائب ليس لديه سوى راتبه الوحيد ليصرح به:

    • نحسب عدد حصص الموظف nbParts=nbEnfants/2 +1 إذا كان غير متزوج، nbEnfants/2+2 إذا كان متزوجًا، حيث nbEnfants هو عدد أطفاله.
    • وإذا كان لديه ثلاثة أطفال على الأقل، يحصل على نصف حصة إضافية
    • يُحسب دخله الخاضع للضريبة R=0.72*S حيث S هو راتبه السنوي
    • يُحسب معامل الأسرة الخاص به QF = R/nbParts

    تم تعريف فئتين لحساب الضريبة: impots و impotsJDBC. يتم إنشاء مثيل للفئة impots باستخدام بيانات شرائح الضرائب التي تم تمريرها كمعلمات في مصفوفات:

    // إنشاء فئة الضرائب

public class impots{

  // البيانات اللازمة لحساب الضريبة
    // تأتي من مصدر خارجي

  protected double[] limites=null;
  protected double[] coeffR=null;
  protected double[] coeffN=null;

    // منشئ فارغ
  protected impots(){}

    // منشئ
  public impots(double[] LIMITES, double[] COEFFR, double[] COEFFN) throws Exception{
        // يتم التحقق من أن الجداول الثلاثة لها نفس الحجم
    boolean OK=LIMITES.length==COEFFR.length && LIMITES.length==COEFFN.length;
    if (! OK) throw new Exception ("Les 3 tableaux fournis n'ont pas la même taille("+
      LIMITES.length+","+COEFFR.length+","+COEFFN.length+")");
    // كل شيء على ما يرام
    this.limites=LIMITES;
    this.coeffR=COEFFR;
    this.coeffN=COEFFN;
  }//المُصنِّع

    // حساب الضريبة
  public long calculer(boolean marié, int nbEnfants, int salaire){
        // حساب عدد الحصص
    double nbParts;
    if (marié) nbParts=(double)nbEnfants/2+2;
    else nbParts=(double)nbEnfants/2+1;
    if (nbEnfants>=3) nbParts+=0.5;
                // حساب الدخل الخاضع للضريبة و«الناتج العائلي»
        double revenu=0.72*salaire;
        double QF=revenu/nbParts;
                // حساب الضريبة
        limites[limites.length-1]=QF+1;
        int i=0;
        while(QF>limites[i]) i++;
        // إرجاع النتيجة
        return (long)(revenu*coeffR[i]-nbParts*coeffN[i]);
    }//الحساب
}//التصنيف

    تُشتق الفئة impotsJDBC من الفئة impots السابقة. يتم إنشاء مثيل للفئة impotsJDBC باستخدام بيانات شرائح الضرائب المخزنة في قاعدة بيانات. يتم تمرير المعلومات اللازمة للوصول إلى قاعدة البيانات هذه كمعلمات إلى مُنشئ الفئة:

    // الحزم المستوردة
import java.sql.*;
import java.util.*;

public class impotsJDBC extends impots{
  // إضافة مُنشئ يسمح بالإنشاء
    // الجداول limits و coeffr و coeffn انطلاقًا من الجدول
    // الضرائب من قاعدة بيانات
  public impotsJDBC(String dsnIMPOTS, String userIMPOTS, String mdpIMPOTS)
      throws SQLException,ClassNotFoundException{

    // dsnIMPOTS: اسم قاعدة البيانات DSN
        // userIMPOTS، mdpIMPOTS: اسم المستخدم/كلمة المرور للوصول إلى قاعدة البيانات

        // جداول البيانات
    ArrayList aLimites=new ArrayList();
    ArrayList aCoeffR=new ArrayList();
    ArrayList aCoeffN=new ArrayList();

    // الاتصال بقاعدة البيانات
    Class.forName("sun.jdbc.odbc.JdbcOdbcDriver");
    Connection connect=DriverManager.getConnection("jdbc:odbc:"+dsnIMPOTS,userIMPOTS,mdpIMPOTS);
    // إنشاء كائن Statement
    Statement S=connect.createStatement();
    // استعلام SELECT
    String select="select limites, coeffr, coeffn from impots";
    // تنفيذ الاستعلام
    ResultSet RS=S.executeQuery(select);
    while(RS.next()){
      // معالجة السطر الحالي
      aLimites.add(RS.getString("limites"));
      aCoeffR.add(RS.getString("coeffr"));
      aCoeffN.add(RS.getString("coeffn"));
    }// السطر التالي
        // إغلاق الموارد
    RS.close();
    S.close();
    connect.close();
        // نقل البيانات إلى مصفوفات محددة
    int n=aLimites.size();
    limites=new double[n];
    coeffR=new double[n];
    coeffN=new double[n];
    for(int i=0;i<n;i++){
      limites[i]=Double.parseDouble((String)aLimites.get(i));
      coeffR[i]=Double.parseDouble((String)aCoeffR.get(i));
      coeffN[i]=Double.parseDouble((String)aCoeffN.get(i));
    }//for
  }//منشئ
}//فئة

    public void calculerImpots(URL urlImpots,boolean marié, int nbEnfants, int salaire)
          throws Exception{

                // urlImpots: URL من مصلحة الضرائب

        String query="?"+"optMarie="+(marié ? "oui":"non")+"&txtEnfants="+nbEnfants+"&txtSalaire="+salaire;

        ArrayList listeSimulations=new ArrayList();

                résultat=ptnErreur.matcher(ligne);
                if(résultat.find()){
                    // رسالة خطأ

                    return listeSimulations;
                }//إذا
            }//إذا
                        // يتم مقارنة السطر بنموذج المحاكاة
            résultat=ptnSimulation.matcher(ligne);
            if(résultat.find()){

    • الإصدار 2: تمت إضافة بعض القدرات من جانب المتصفح عن طريق إدراج نصوص جافا سكريبت في المستند HTML الذي يتم تحميله بواسطة المتصفح. ويتحقق المتصفح من صحة معلمات النموذج.
    • الإصدار 3: نسمح للخدمة بتذكر المحاكاة المختلفة التي أجراها العميل من خلال إدارة جلسة عمل. يتم تعديل الواجهة HTML وفقًا لذلك لعرض هذه المحاكاة.
    • الإصدار 4: أصبح العميل الآن تطبيقًا رسوميًا مستقلًا. وهذا يسمح لنا بالعودة إلى كتابة برامج العملاء على الويب.
    • الإصدار 5: يتحول العميل إلى تطبيق Java صغير (applet). وبذلك يصبح لدينا تطبيق عميل-خادم مبرمج بالكامل بلغة Java سواء من جانب الخادم أو من جانب العميل.

    في هذه المرحلة، يمكننا إبداء بعض الملاحظات:

    • تسمح الإصدارات من 1 إلى 3 باستخدام متصفحات لا تمتلك أي قدرات أخرى سوى القدرة على تنفيذ نصوص جافا سكريبت. وتجدر الإشارة إلى أن المستخدم لديه دائمًا إمكانية منع تنفيذ هذه البرامج النصية. وعندئذٍ لن يعمل التطبيق إلا جزئيًا في الإصدار 1 (لن تعمل خيار «حذف») ولن يعمل على الإطلاق في الإصدارين 2 و3 (لن تعمل خيارات «حذف» و«حساب»). وقد يكون من المفيد توفير إصدار من الخدمة لا يستخدم نصوص جافا سكريبت.
    • يتطلب الإصدار 4 أن يكون جهاز العميل مزودًا بآلة افتراضية Java 2.
    • يتطلب الإصدار 5 أن يكون لدى جهاز العميل متصفح مزود بجهاز افتراضي Java 2.

    عند كتابة خدمة ويب، يجب أن نسأل أنفسنا عن أنواع العملاء التي نستهدفها. إذا أردنا استهداف أكبر عدد ممكن من العملاء، فسنكتب تطبيقًا لا يرسل إلى المتصفحات سوى HTML (بدون جافا سكريبت أو أبلت). أما إذا كنا نعمل ضمن شبكة إنترانت ونتحكم في تكوين أجهزة المستخدمين فيها، فيمكننا عندئذٍ أن نكون أكثر صرامةً فيما يتعلق بمتطلبات العميل، وقد تصبح الإصدار 5 السابق مقبولاً في هذه الحالة.

    الإصداران 4 و5 هما عميلان ويب يستخرجان المعلومات التي يحتاجانها من تدفق البيانات HTML الذي يرسله الخادم. في كثير من الأحيان، لا نتمتع بالسيطرة على هذا التدفق. وهذا هو الحال عندما نكتب عميلاً لخدمة ويب موجودة على الشبكة ويديرها شخص آخر. لنأخذ مثالاً. لنفترض أن خدمة محاكاة حساب الضرائب الخاصة بنا قد تمت برمجتها بواسطة شركة X. حالياً، ترسل الخدمة المحاكاة في جدول HTML، ويستفيد عميلنا من هذه الحقيقة لاستردادها. وبذلك، يقارن كل سطر من رد الخادم بالتعبير النمطي:

                    // نموذج لسطر واحد من جدول المحاكاة
        Pattern ptnSimulation=Pattern.compile("<tr>\\s*<td>(.*?)</td>\\s*<td>(.*?)</td>\\s*<td>(.*?)</td>\\s*<td>(.*?)</td>\\s*</tr>");

    لنفترض الآن أن مطور التطبيق قام بتغيير الشكل المرئي للاستجابة، بحيث لم يعد يضع المحاكاة في مصفوفة بل في قائمة بالشكل التالي:

    Résultat des simulations
<ul>
    <li>oui,2,200000,22504
  <li>non,2,200000,33388
</ul>

    في هذه الحالة، سيتعين إعادة كتابة عميل الويب الخاص بنا. وهذا هو التهديد الدائم الذي يواجه عملاء الويب للتطبيقات التي لا نسيطر عليها بأنفسنا. يمكن لـ XML أن يقدم حلاً لهذه المشكلة:

    • بدلاً من إنشاء HTML، ستقوم خدمة المحاكاة بإنشاء XML. في مثالنا، قد يكون ذلك 
    <simulations>
    <entetes marie="marié" enfants="enfants" salaire="salaire" impot="impôt"/>
  <simulation marie="oui" enfants="2" salaire="200000" impot="22504" />
  <simulation marie="non" enfants="2" salaire="200000" impot="33388" />
</simulations>
    • يمكن ربط ورقة أنماط بهذا الرد لتوجيه المتصفحات إلى الشكل المرئي الذي يجب أن يتخذه هذا الرد XML
    • ستتجاهل برامج الويب هذه ورقة الأنماط وستسترد المعلومات مباشرةً من تدفق XML في الرد

    إذا رغب مصمم الخدمة في تعديل الشكل المرئي للنتائج التي يقدمها، فسيقوم بتعديل ورقة الأنماط وليس ملف XML. وبفضل ورقة الأنماط، ستعرض المتصفحات الشكل المرئي الجديد ولن يتعين تعديل عملاء الويب المبرمجين. وبالتالي، يمكن كتابة إصدارات جديدة من خدمة المحاكاة الخاصة بنا:
